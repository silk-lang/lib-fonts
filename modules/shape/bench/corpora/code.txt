fn firstGlyphAt(state: &Shaping, item: Item, cluster: usize) -> usize {
  let mut low = item.glyphStart
  let mut high = item.glyphEnd
  while low < high {
    let middle = low + (high - low) / 2
    let at = u32.toUsize(state.raw.get(middle).cluster)
    if at < cluster { low = middle + 1 } else { high = middle }
  }
  return low
}
